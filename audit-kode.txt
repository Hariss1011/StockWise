<?php

namespace App\Services;

use App\Models\Product;
use App\Models\StockIn;
use App\Models\StockOut;
use App\Models\StockAdjustment;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\ValidationException;

class StockService
{
    public function currentStock(Product $product): int
    {
        $stockIn = $product->stockIns()->sum('quantity');
        $stockOut = $product->stockOuts()->sum('quantity');
        $adjustment = $product->stockAdjustments()->sum('quantity_change');

        return (int) $product->initial_stock
            + (int) $stockIn
            - (int) $stockOut
            + (int) $adjustment;
    }

    public function recordIn(array $data): StockIn
    {
        return DB::transaction(function () use ($data) {
            Product::whereKey($data['product_id'])
                ->lockForUpdate()
                ->firstOrFail();

            return StockIn::create($data);
        });
    }

    public function recordOut(array $data): StockOut
    {
        return DB::transaction(function () use ($data) {
            $product = Product::whereKey($data['product_id'])
                ->lockForUpdate()
                ->firstOrFail();

            if ($this->currentStock($product) < $data['quantity']) {
                throw ValidationException::withMessages([
                    'quantity' => 'Stok tidak mencukupi untuk transaksi ini.',
                ]);
            }

            return StockOut::create($data);
        });
    }

    public function recordAdjustment(array $data): StockAdjustment
    {
        return DB::transaction(function () use ($data) {
            $product = Product::whereKey($data['product_id'])
                ->lockForUpdate()
                ->firstOrFail();

            $newStock = $this->currentStock($product)
                + $data['quantity_change'];

            if ($newStock < 0) {
                throw ValidationException::withMessages([
                    'quantity_change' => 'Penyesuaian ditolak karena stok akan menjadi negatif.',
                ]);
            }

            return StockAdjustment::create($data);
        });
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\Product;
use App\Models\StockIn;
use App\Models\StockOut;
use App\Models\StockAdjustment;
use App\Services\StockService;
use Illuminate\Http\Request;

class StockTransactionController extends Controller
{
    public function __construct(
        private StockService $stockService
    ) {}

    public function index()
    {
        $stockIns = StockIn::with('product')
            ->latest('transaction_date')
            ->latest('id')
            ->paginate(10, ['*'], 'masuk');

        $stockOuts = StockOut::with('product')
            ->latest('transaction_date')
            ->latest('id')
            ->paginate(10, ['*'], 'keluar');

        $adjustments = StockAdjustment::with('product')
            ->latest('adjustment_date')
            ->latest('id')
            ->paginate(10, ['*'], 'penyesuaian');

        return view('transactions.index', compact(
            'stockIns',
            'stockOuts',
            'adjustments'
        ));
    }

    public function createIn()
    {
        return $this->form('in');
    }

    public function storeIn(Request $request)
    {
        $data = $request->validate([
            'product_id' => ['required', 'exists:products,id'],
            'reference_number' => ['nullable', 'string', 'max:100'],
            'quantity' => ['required', 'integer', 'min:1'],
            'unit_cost' => ['nullable', 'numeric', 'min:0'],
            'transaction_date' => ['required', 'date'],
            'notes' => ['nullable', 'string'],
        ]);

        $this->stockService->recordIn($data);

        return redirect()->route('transactions.index')
            ->with('success', 'Barang masuk berhasil dicatat.');
    }

    public function createOut()
    {
        return $this->form('out');
    }

    public function storeOut(Request $request)
    {
        $data = $request->validate([
            'product_id' => ['required', 'exists:products,id'],
            'reference_number' => ['nullable', 'string', 'max:100'],
            'quantity' => ['required', 'integer', 'min:1'],
            'recipient' => ['nullable', 'string', 'max:150'],
            'transaction_date' => ['required', 'date'],
            'notes' => ['nullable', 'string'],
        ]);

        $this->stockService->recordOut($data);

        return redirect()->route('transactions.index')
            ->with('success', 'Barang keluar berhasil dicatat.');
    }

    public function createAdjustment()
    {
        return $this->form('adjustment');
    }

    public function storeAdjustment(Request $request)
    {
        $data = $request->validate([
            'product_id' => ['required', 'exists:products,id'],
            'quantity_change' => [
                'required',
                'integer',
                'not_in:0',
            ],
            'reason' => ['required', 'string', 'max:100'],
            'adjustment_date' => ['required', 'date'],
            'notes' => ['nullable', 'string'],
        ]);

        $this->stockService->recordAdjustment($data);

        return redirect()->route('transactions.index')
            ->with('success', 'Penyesuaian stok berhasil dicatat.');
    }

    private function form(string $type)
    {
        return view('transactions.form', [
            'type' => $type,
            'products' => Product::orderBy('name')->get(),
        ]);
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\Category;
use App\Models\Product;
use App\Models\Supplier;
use Illuminate\Http\Request;
use Illuminate\Validation\Rule;

class ProductController extends Controller
{

    public function index()
    {
        $products = Product::with(['category', 'supplier'])
            ->withSum('stockIns', 'quantity')
            ->withSum('stockOuts', 'quantity')
            ->withSum('stockAdjustments', 'quantity_change')
            ->latest()
            ->paginate(10);

        return view('products.index', compact('products'));
    }


    public function create()
    {
        return view('products.form', [
            'product' => new Product(),
            'categories' => Category::orderBy('name')->get(),
            'suppliers' => Supplier::orderBy('name')->get(),
            'isEdit' => false,
        ]);
    }

    public function store(Request $request)
    {
        $validated = $request->validate([
            'category_id' => ['required', 'exists:categories,id'],
            'supplier_id' => ['nullable', 'exists:suppliers,id'],
            'sku' => ['required', 'string', 'max:50', 'unique:products,sku'],
            'name' => ['required', 'string', 'max:150'],
            'description' => ['nullable', 'string'],
            'unit' => ['required', 'string', 'max:30'],
            'purchase_price' => ['required', 'numeric', 'min:0'],
            'selling_price' => ['required', 'numeric', 'min:0'],
            'initial_stock' => ['required', 'integer', 'min:0'],
            'minimum_stock' => ['required', 'integer', 'min:0'],
        ]);

        Product::create($validated);

        return redirect()->route('products.index')
            ->with('success', 'Produk berhasil ditambahkan.');
    }

    public function edit(Product $product)
    {
        return view('products.form', [
            'product' => $product,
            'categories' => Category::orderBy('name')->get(),
            'suppliers' => Supplier::orderBy('name')->get(),
            'isEdit' => true,
        ]);
    }

    public function update(Request $request, Product $product)
    {
        $validated = $request->validate([
            'category_id' => ['required', 'exists:categories,id'],
            'supplier_id' => ['nullable', 'exists:suppliers,id'],
            'sku' => [
                'required',
                'string',
                'max:50',
                Rule::unique('products', 'sku')->ignore($product->id),
            ],
            'name' => ['required', 'string', 'max:150'],
            'description' => ['nullable', 'string'],
            'unit' => ['required', 'string', 'max:30'],
            'purchase_price' => ['required', 'numeric', 'min:0'],
            'selling_price' => ['required', 'numeric', 'min:0'],
            'minimum_stock' => ['required', 'integer', 'min:0'],
        ]);

        // Stok awal tidak diubah saat mengedit produk.
        $product->update($validated);

        return redirect()->route('products.index')
            ->with('success', 'Produk berhasil diperbarui.');
    }

    public function destroy(Product $product)
    {
        if (
            $product->stockIns()->exists() ||
            $product->stockOuts()->exists() ||
            $product->stockAdjustments()->exists()
        ) {
            return back()->with(
                'error',
                'Produk memiliki riwayat transaksi sehingga tidak bisa dihapus.'
            );
        }

        $product->delete();

        return back()->with('success', 'Produk berhasil dihapus.');
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\Category;
use App\Models\Product;
use App\Models\StockIn;
use App\Models\StockOut;
use App\Models\StockAdjustment;
use Illuminate\Http\Request;
use Illuminate\Pagination\LengthAwarePaginator;
use Symfony\Component\HttpFoundation\StreamedResponse;

class ReportController extends Controller
{
    public function stock(Request $request)
    {
        $filters = $request->validate([
            'search' => ['nullable', 'string', 'max:100'],
            'category_id' => ['nullable', 'integer', 'exists:categories,id'],
            'status' => ['nullable', 'in:all,low,normal'],
        ]);

        $products = Product::with('category')
            ->withSum('stockIns', 'quantity')
            ->withSum('stockOuts', 'quantity')
            ->withSum('stockAdjustments', 'quantity_change')
            ->when($request->filled('search'), function ($query) use ($request) {
                $search = $request->input('search');

                $query->where(function ($q) use ($search) {
                    $q->where('name', 'like', "%{$search}%")
                        ->orWhere('sku', 'like', "%{$search}%");
                });
            })
            ->when($request->filled('category_id'), function ($query) use ($request) {
                $query->where('category_id', $request->input('category_id'));
            })
            ->orderBy('name')
            ->get()
            ->map(function ($product) {
                $product->current_stock =
                    (int) $product->initial_stock
                    + (int) ($product->stock_ins_sum_quantity ?? 0)
                    - (int) ($product->stock_outs_sum_quantity ?? 0)
                    + (int) ($product->stock_adjustments_sum_quantity_change ?? 0);

                $product->inventory_value =
                    $product->current_stock * (float) $product->purchase_price;

                return $product;
            });

        if (($filters['status'] ?? 'all') === 'low') {
            $products = $products->filter(
                fn($product) => $product->current_stock <= $product->minimum_stock
            );
        } elseif (($filters['status'] ?? 'all') === 'normal') {
            $products = $products->filter(
                fn($product) => $product->current_stock > $product->minimum_stock
            );
        }

        $summary = [
            'product_count' => $products->count(),
            'total_units' => $products->sum('current_stock'),
            'inventory_value' => $products->sum('inventory_value'),
        ];

        $perPage = 15;
        $page = LengthAwarePaginator::resolveCurrentPage();
        $paginatedProducts = new LengthAwarePaginator(
            $products->forPage($page, $perPage)->values(),
            $products->count(),
            $perPage,
            $page,
            [
                'path' => $request->url(),
                'query' => $request->query(),
            ]
        );

        return view('reports.stock', [
            'products' => $paginatedProducts,
            'categories' => Category::orderBy('name')->get(),
            'summary' => $summary,
        ]);
    }

    public function exportStock(Request $request): StreamedResponse
    {
        $filters = $request->validate([
            'search' => ['nullable', 'string', 'max:100'],
            'category_id' => ['nullable', 'integer', 'exists:categories,id'],
            'status' => ['nullable', 'in:all,low,normal'],
        ]);

        $products = Product::with('category')
            ->withSum('stockIns', 'quantity')
            ->withSum('stockOuts', 'quantity')
            ->withSum('stockAdjustments', 'quantity_change')
            ->when(!empty($filters['search']), function ($query) use ($filters) {
                $search = $filters['search'];

                $query->where(function ($q) use ($search) {
                    $q->where('name', 'like', "%{$search}%")
                        ->orWhere('sku', 'like', "%{$search}%");
                });
            })
            ->when(!empty($filters['category_id']), function ($query) use ($filters) {
                $query->where('category_id', $filters['category_id']);
            })
            ->orderBy('name')
            ->get()
            ->map(function ($product) {
                $product->current_stock =
                    (int) $product->initial_stock
                    + (int) ($product->stock_ins_sum_quantity ?? 0)
                    - (int) ($product->stock_outs_sum_quantity ?? 0)
                    + (int) ($product->stock_adjustments_sum_quantity_change ?? 0);

                $product->inventory_value =
                    $product->current_stock * (float) $product->purchase_price;

                return $product;
            });

        $status = $filters['status'] ?? 'all';

        if ($status === 'low') {
            $products = $products->filter(
                fn($product) => $product->current_stock <= $product->minimum_stock
            );
        } elseif ($status === 'normal') {
            $products = $products->filter(
                fn($product) => $product->current_stock > $product->minimum_stock
            );
        }

        $filename = 'laporan-stok-' . now()->format('Y-m-d-His') . '.csv';

        return response()->streamDownload(function () use ($products) {
            $handle = fopen('php://output', 'w');

            // BOM membantu Excel mengenali encoding UTF-8.
            fwrite($handle, "\xEF\xBB\xBF");

            fputcsv($handle, [
                'SKU',
                'Nama Produk',
                'Kategori',
                'Stok Awal',
                'Barang Masuk',
                'Barang Keluar',
                'Penyesuaian',
                'Stok Akhir',
                'Stok Minimum',
                'Harga Beli',
                'Nilai Persediaan',
                'Status Stok',
            ]);

            foreach ($products as $product) {
                $stockIn = (int) ($product->stock_ins_sum_quantity ?? 0);
                $stockOut = (int) ($product->stock_outs_sum_quantity ?? 0);
                $adjustment = (int) (
                    $product->stock_adjustments_sum_quantity_change ?? 0
                );

                fputcsv($handle, [
                    $this->safeCsvText($product->sku),
                    $this->safeCsvText($product->name),
                    $this->safeCsvText($product->category?->name ?? '-'),
                    $product->initial_stock,
                    $stockIn,
                    $stockOut,
                    $adjustment,
                    $product->current_stock,
                    $product->minimum_stock,
                    $product->purchase_price,
                    $product->inventory_value,
                    $product->current_stock <= $product->minimum_stock
                        ? 'Stok rendah'
                        : 'Normal',
                ]);
            }

            fclose($handle);
        }, $filename, [
            'Content-Type' => 'text/csv; charset=UTF-8',
        ]);
    }

    public function exportTransactions(Request $request): StreamedResponse
    {
        $filters = $request->validate([
            'date_from' => ['nullable', 'date'],
            'date_to' => [
                'nullable',
                'date',
                'after_or_equal:date_from',
            ],
            'type' => ['nullable', 'in:all,in,out,adjustment'],
        ]);

        $dateFrom = $filters['date_from'] ?? null;
        $dateTo = $filters['date_to'] ?? null;
        $type = $filters['type'] ?? 'all';

        $stockIns = collect();
        $stockOuts = collect();
        $adjustments = collect();

        if ($type === 'all' || $type === 'in') {
            $stockIns = StockIn::with('product')
                ->when(
                    $dateFrom,
                    fn($q) =>
                    $q->whereDate('transaction_date', '>=', $dateFrom)
                )
                ->when(
                    $dateTo,
                    fn($q) =>
                    $q->whereDate('transaction_date', '<=', $dateTo)
                )
                ->orderBy('transaction_date')
                ->orderBy('id')
                ->get();
        }

        if ($type === 'all' || $type === 'out') {
            $stockOuts = StockOut::with('product')
                ->when(
                    $dateFrom,
                    fn($q) =>
                    $q->whereDate('transaction_date', '>=', $dateFrom)
                )
                ->when(
                    $dateTo,
                    fn($q) =>
                    $q->whereDate('transaction_date', '<=', $dateTo)
                )
                ->orderBy('transaction_date')
                ->orderBy('id')
                ->get();
        }

        if ($type === 'all' || $type === 'adjustment') {
            $adjustments = StockAdjustment::with('product')
                ->when(
                    $dateFrom,
                    fn($q) =>
                    $q->whereDate('adjustment_date', '>=', $dateFrom)
                )
                ->when(
                    $dateTo,
                    fn($q) =>
                    $q->whereDate('adjustment_date', '<=', $dateTo)
                )
                ->orderBy('adjustment_date')
                ->orderBy('id')
                ->get();
        }

        $filename = 'laporan-transaksi-' . now()->format('Y-m-d-His') . '.csv';

        return response()->streamDownload(function () use (
            $stockIns,
            $stockOuts,
            $adjustments
        ) {
            $handle = fopen('php://output', 'w');

            fwrite($handle, "\xEF\xBB\xBF");

            fputcsv($handle, [
                'Jenis Transaksi',
                'Tanggal',
                'Nomor Referensi',
                'SKU',
                'Nama Produk',
                'Jumlah',
                'Satuan',
                'Detail',
                'Catatan',
            ]);

            foreach ($stockIns as $item) {
                fputcsv($handle, [
                    'Barang Masuk',
                    $item->transaction_date->format('Y-m-d'),
                    $this->safeCsvText($item->reference_number ?? ''),
                    $this->safeCsvText($item->product->sku),
                    $this->safeCsvText($item->product->name),
                    $item->quantity,
                    $this->safeCsvText($item->product->unit),
                    $item->unit_cost === null
                        ? ''
                        : 'Biaya/unit: ' . $item->unit_cost,
                    $this->safeCsvText($item->notes ?? ''),
                ]);
            }

            foreach ($stockOuts as $item) {
                fputcsv($handle, [
                    'Barang Keluar',
                    $item->transaction_date->format('Y-m-d'),
                    $this->safeCsvText($item->reference_number ?? ''),
                    $this->safeCsvText($item->product->sku),
                    $this->safeCsvText($item->product->name),
                    $item->quantity,
                    $this->safeCsvText($item->product->unit),
                    'Penerima: ' . $this->safeCsvText($item->recipient ?? ''),
                    $this->safeCsvText($item->notes ?? ''),
                ]);
            }

            foreach ($adjustments as $item) {
                fputcsv($handle, [
                    'Penyesuaian',
                    $item->adjustment_date->format('Y-m-d'),
                    '',
                    $this->safeCsvText($item->product->sku),
                    $this->safeCsvText($item->product->name),
                    $item->quantity_change,
                    $this->safeCsvText($item->product->unit),
                    'Alasan: ' . $this->safeCsvText($item->reason),
                    $this->safeCsvText($item->notes ?? ''),
                ]);
            }

            fclose($handle);
        }, $filename, [
            'Content-Type' => 'text/csv; charset=UTF-8',
        ]);
    }

    /**
     * Mencegah teks yang diawali karakter formula
     * diperlakukan sebagai formula saat dibuka di spreadsheet.
     */
    private function safeCsvText(?string $value): string
    {
        $value = (string) $value;

        if (preg_match('/^[\s]*[=+\-@]/u', $value)) {
            return "'" . $value;
        }

        return $value;
    }
}
